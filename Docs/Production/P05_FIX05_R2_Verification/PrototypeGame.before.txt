        private void Awake()
        {
            if (theme == null) theme = ScriptableObject.CreateInstance<PrototypeUITheme>();
            game = useFixedMock ? (IPrototypeUIAdapter)new FixedPrototypeUIAdapter() : new MemoryPrototypeUIAdapter();
            font = Font.CreateDynamicFontFromOSFont(new[] { "Microsoft YaHei", "SimHei", "Arial" }, 28);
            if (HasArt) font = Font.CreateDynamicFontFromOSFont(new[] { "SimSun", "STSong", "Microsoft YaHei" }, 28);
            titleFont = Font.CreateDynamicFontFromOSFont(new[] { "STZhongsong", "SimSun", "Microsoft YaHei" }, 64);
            if (FindObjectOfType<Camera>() == null)
            {
                var cameraObject = new GameObject("PrototypeCamera", typeof(Camera));
                cameraObject.transform.SetParent(transform, false);
                Camera camera = cameraObject.GetComponent<Camera>();
                camera.clearFlags = CameraClearFlags.SolidColor;
                camera.backgroundColor = background;
                camera.orthographic = true;
            }
            if (signatureFont == null) signatureFont = Resources.Load<Font>("Fonts/LXGWWenKai-Regular");
            if (signatureFont != null) handwriting = signatureFont;
            else
            {
                string[] installed = Font.GetOSInstalledFontNames();
                string family = Array.Find(installed, name => string.Equals(name, "STXingkai", StringComparison.OrdinalIgnoreCase) || name == "华文行楷");
                handwriting = Font.CreateDynamicFontFromOSFont(family ?? "KaiTi", 58);
                Debug.Log("[P03] 签名系统字体：" + (family ?? "KaiTi"));
            }
            summonAudio = gameObject.AddComponent<SummonAudio>(); summonAudio.Configure(feelConfig);
            BuildInterface();
            ShowSummon();
        }

        private void Update()
        {
            if (FlowStage == SummonFlowStage.AwaitingConfirm) fastReveal.TickIdle(Time.unscaledTime);
            if (Input.GetKeyDown(KeyCode.Escape)) { HandleEscape(); return; }
            if (currentScreen == "Reveal" && (Input.GetKeyDown(KeyCode.Space) || Input.GetKeyDown(KeyCode.Return)))
            {if(FlowStage==SummonFlowStage.CrystalCharging)HitCrystal();else AdvancePresentation();}
        }

        public void HandleEscape()
        {
            if (FlowStage == SummonFlowStage.Signing || FlowStage == SummonFlowStage.AutoSigning) { CancelSigning(); return; }
            if (currentScreen == "Reveal") { SkipPresentation(); return; }
            if (modal.gameObject.activeSelf) CloseRules();
            else if (currentScreen == "Detail")
            { if (detailChrome.alpha < 0.5f) ToggleDetailUI(); else ShowScreen(detailReturn); }
            else if (currentScreen == "Collection") ShowSummon();
        }

        private void Summon(int count) { RequestSummon(count); }

        public void RequestSummon(int count)
        {
            if (busy || modal.gameObject.activeSelf || (count != 1 && count != 10) || game.Currency < count) return;
            busy = true; pendingSummonCount = count; RevealIndex = -1;
            signaturePad.ClearStroke(); signaturePad.AcceptInput = true; autoSignatureText.text = "";
            autoSignatureMask.anchorMax = Vector2.one;
            SetFlowStage(SummonFlowStage.Signing);
            ShowScreen("Signing"); UpdateResources();
        }

        public void CompleteHandwrittenSignature()
        {
            if (FlowStage != SummonFlowStage.Signing || signaturePad.Points.Count == 0) return;
            CommitContract();
        }
        public void SkipSigning()
        {
            if (FlowStage != SummonFlowStage.Signing) return;
            SetFlowStage(SummonFlowStage.AutoSigning);
            signaturePad.AcceptInput = false; signaturePad.ClearStroke();
            autoSignatureText.text = string.IsNullOrWhiteSpace(playerDisplayName) ? "召唤者" : playerDisplayName;
            signingRoutine = StartCoroutine(AutoSign());
        }
        private IEnumerator AutoSign()
        {
            float elapsed = 0;
            while (elapsed < autoSignatureSeconds)
            {
                elapsed += Time.unscaledDeltaTime;
                autoSignatureMask.anchorMax = new Vector2(Mathf.Clamp01(elapsed / Mathf.Max(0.01f, autoSignatureSeconds)), 1);
                // 保持文本自身宽度，遮罩仅负责横向显露。
                autoSignatureText.rectTransform.SetSizeWithCurrentAnchors(RectTransform.Axis.Horizontal, signaturePad.rectTransform.rect.width);
                yield return null;
            }
            autoSignatureMask.anchorMax = Vector2.one;
            signingRoutine = null;
            CommitContract();
        }
        public void CancelSigning()
        {
            if (FlowStage != SummonFlowStage.Signing && FlowStage != SummonFlowStage.AutoSigning) return;
            if (signingRoutine != null) StopCoroutine(signingRoutine);
            signingRoutine = null; pendingSummonCount = 0; busy = false; signaturePad.AcceptInput = false;
            signaturePad.ClearStroke(); autoSignatureText.text = "";
            SetFlowStage(SummonFlowStage.Idle); ShowSummon();
        }
        private void CommitContract()
        {
            if ((FlowStage != SummonFlowStage.Signing && FlowStage != SummonFlowStage.AutoSigning) || pendingSummonCount == 0) return;
            SetFlowStage(SummonFlowStage.Commit); signaturePad.AcceptInput = false;
            int count = pendingSummonCount; pendingSummonCount = 0;
            lastResults = game.Draw(count); CommittedSummonRequests++;
            UpdateResources(); revealRoutine = StartCoroutine(Reveal());
        }

        // Editor/开发版指定批次：不调用 Draw，不改收藏。
        public void PlayBatchForDebug(List<PrototypePull> results)
        {
            if ((!Application.isEditor && !Debug.isDebugBuild) || busy || results == null || results.Count == 0) return;
            busy = true; lastResults = new List<PrototypePull>(results);
            revealRoutine = StartCoroutine(Reveal());
        }
        private IEnumerator Reveal()
        {
            ShowScreen("Reveal"); refillButton.interactable = false; RevealIndex = -1;
            cardStage.gameObject.SetActive(false); ceremonyStage.gameObject.SetActive(true);
            revealCount.gameObject.SetActive(false); confirmCardButton.gameObject.SetActive(false);
            SetFlowStage(SummonFlowStage.Ceremony); finishCurrentPhase = false;
            CardRarity highest = CardRarity.R; foreach (PrototypePull result in lastResults) if (result.Card.Rarity > highest) highest = result.Card.Rarity;
            bool crystal=crystalView!=null;
            float prelude=crystal?feelConfig.preludeSeconds:ceremonySeconds;
            ceremonyView.Begin(signaturePad, autoSignatureText.text, highest, prelude);
            float elapsed = 0;
            while (elapsed < prelude && !finishCurrentPhase)
            { elapsed += Time.unscaledDeltaTime; ceremonyView.Animate(elapsed / Mathf.Max(0.01f, prelude)); yield return null; }
            ceremonyView.Finish(); ceremonyStage.gameObject.SetActive(false);
            if(crystal)
            {
                crystalView.Begin(highest);summonAudio.CrystalWait();SetFlowStage(SummonFlowStage.CrystalCharging);
                while(crystalView.Energy<1){crystalView.Tick();yield return null;}
                SetFlowStage(SummonFlowStage.CrystalBreaking);summonAudio.FinalDeclaration();
                for(float t=0;t<feelConfig.finalChargeSeconds;t+=Time.unscaledDeltaTime){crystalView.Terminal(t);yield return null;}
                summonAudio.CrystalBreak();crystalView.Burst(0);
            }
            cardStage.gameObject.SetActive(true);
            revealCount.gameObject.SetActive(true);
            for (int i = 0; i < lastResults.Count; i++)
            {
                RevealIndex = i; PrototypePull pull = lastResults[i];
                SetFlowStage(SummonFlowStage.Revealing); finishCurrentPhase = false; confirmedCard = false;
                confirmCardButton.gameObject.SetActive(false);
                revealCount.text = (i + 1) + "/" + lastResults.Count;
                PrototypePortrait binding = portraits.Find(item => item.characterId == pull.Card.Id);
                Sprite portrait = pull.Card.Id == "SSR001" && theme.summonCharacter != null ? theme.summonCharacter : binding == null ? null : binding.portrait;
                revealInputReadyFrame=Time.frameCount+2;
                fastReveal.Begin(pull.Card, portrait, pull.IsNew);
                cardStage.Find("RevealArtworkSlot/UnknownSilhouette").GetComponent<Image>().enabled = false;
                revealSilhouette.enabled = portrait == null;
                float duration = pull.Card.Rarity == CardRarity.UR ? urRevealSeconds : pull.Card.Rarity == CardRarity.SSR ? ssrRevealSeconds
                    : pull.Card.Rarity == CardRarity.SR ? srRevealSeconds : normalRevealSeconds;
                elapsed = 0;
                while (elapsed < duration && !finishCurrentPhase)
                { elapsed += Time.unscaledDeltaTime; fastReveal.Animate(elapsed / Mathf.Max(0.01f, duration));
                  if(crystal&&i==0){crystalView.Burst(elapsed);if(elapsed>=feelConfig.whiteSeconds)crystalView.Clear();}yield return null; }
                if(crystal)crystalView.Clear();
                fastReveal.Animate(1); fastReveal.Finish();
                SetFlowStage(SummonFlowStage.AwaitingConfirm); confirmationReadyFrame = Time.frameCount + 1;
                confirmCardButton.GetComponentInChildren<Text>().text = i + 1 == lastResults.Count ? "查看结果" : "下一张";
                confirmCardButton.gameObject.SetActive(true);
                while (!confirmedCard) yield return null;
                if (i + 1 < lastResults.Count)
                    for (float transitionTime = 0; transitionTime < 0.15f; transitionTime += Time.unscaledDeltaTime) { fastReveal.Transition(transitionTime / 0.15f); yield return null; }
            }
            revealRoutine = null; fastReveal.Finish(); ShowSummary();
        }
        public void AdvancePresentation()
        {
            if(FlowStage==SummonFlowStage.Revealing&&Time.frameCount<revealInputReadyFrame)return;
            if (lastAdvanceFrame == Time.frameCount) return;
            if (FlowStage == SummonFlowStage.Ceremony || FlowStage == SummonFlowStage.Revealing)
            { lastAdvanceFrame = Time.frameCount; finishCurrentPhase = true; }
            else if (FlowStage == SummonFlowStage.AwaitingConfirm && Time.frameCount >= confirmationReadyFrame)
            { lastAdvanceFrame = Time.frameCount; confirmedCard = true; confirmCardButton.gameObject.SetActive(false); summonAudio.Confirm(); }
        }
        public void SkipPresentation()
        {
            if (FlowStage == SummonFlowStage.Signing) { SkipSigning(); return; }
            if (FlowStage != SummonFlowStage.Ceremony && FlowStage != SummonFlowStage.Revealing && FlowStage != SummonFlowStage.AwaitingConfirm && FlowStage != SummonFlowStage.CrystalCharging && FlowStage != SummonFlowStage.CrystalBreaking) return;
            if (revealRoutine != null) StopCoroutine(revealRoutine);
            revealRoutine = null; fastReveal.Finish(); ceremonyView.Finish(); if(crystalView!=null)crystalView.Clear(); summonAudio.Skip(); ShowSummary();
        }

        private RectTransform SpriteSurface(Transform parent, string name, Sprite sprite,
            float x0, float y0, float x1, float y1, bool preserve)
        {
            // Fitter 会接管其对象的锚点，因此先用独立 Slot 保留调用者定义的区域。
            RectTransform slot = preserve && sprite != null ? Rect(parent, name + "Slot", x0, y0, x1, y1) : null;
            RectTransform rect = slot != null ? Surface(slot, name, Color.white, 0, 0, 1, 1)
                : Surface(parent, name, Color.white, x0, y0, x1, y1);
            Image image = rect.GetComponent<Image>();
            image.sprite = sprite;
            if (preserve && sprite != null)
            {
                var fit = rect.gameObject.AddComponent<AspectRatioFitter>();
                fit.aspectMode = AspectRatioFitter.AspectMode.FitInParent;
                fit.aspectRatio = sprite.rect.width / sprite.rect.height;
            }
            return rect;
        }

        private Sprite ButtonSprite(Sprite source)
        {
            Sprite cropped;

            // 使用同一底图裁掉透明留白；不修改 PNG，不做 SpriteSwap。

                SpriteMeshType.FullRect, new Vector4(90, 40, 90, 40));
            cropped.name = source.name + "_VisibleBody";
            croppedButtons.Add(source, cropped);
            return cropped;
        }

        private RectTransform ConceptCard(Transform parent, PrototypeCard card, float x0, float y0, float x1, float y1,

        {

            RectTransform outer = Surface(parent, card.Id, RarityColor(card.Rarity), x0, y0, x1, y1);
            RectTransform window = Surface(outer, "PortraitWindow", background, 0.012f, 0.009f, 0.988f, 0.991f);

            if (binding != null && binding.portrait != null)
            {
                RectTransform image = Rect(window, "Portrait", 0, 0, 1, 1);
                RawImage artwork = image.gameObject.AddComponent<RawImage>();
                artwork.texture = binding.portrait.texture;
                artwork.uvRect = theme.summaryPortraitCrop;
                artwork.raycastTarget = false;
                var fit = image.gameObject.AddComponent<AspectRatioFitter>();
                fit.aspectMode = AspectRatioFitter.AspectMode.FitInParent;
                fit.aspectRatio = binding.portrait.rect.width * theme.summaryPortraitCrop.width
                    / (binding.portrait.rect.height * theme.summaryPortraitCrop.height);
            }
            else
            {
                Silhouette(window);
            }
            Color trim = card.Rarity == CardRarity.R ? new Color(0.28f, 0.64f, 1)
                : card.Rarity == CardRarity.SR ? new Color(0.75f, 0.35f, 1)
                : card.Rarity == CardRarity.SSR ? new Color(1, 0.75f, 0.36f) : new Color(1, 0.65f, 0.30f);
            Image outline = outer.GetComponent<Image>();
            outline.color = trim;
            Surface(outer, "TopTrim", trim, 0, 0.985f, 1, 1);
            Surface(outer, "BottomTrim", trim, 0, 0, 1, 0.015f);
            Title(outer, "◇", 19, -0.025f, 0.94f, 0.09f, 1.04f).color = trim;
            Title(outer, "◇", 19, 0.91f, -0.04f, 1.025f, 0.06f).color = trim;
            RectTransform caption = Surface(outer, "Caption", new Color(0.02f, 0.035f, 0.085f, 0.86f), 0.014f, 0.015f, 0.986f, 0.145f);
            bool compact = parent.name == "CollectionCards";
            Sprite element = theme.ElementIcon(card.Id);
            if (element != null) SpriteSurface(caption, "Element", element, 0.02f, 0.05f, 0.18f, 0.95f, true);
            Title(caption, card.Name, compact ? 22 : 27, element != null ? 0.18f : 0.03f, 0, 0.98f, 1);
            RectTransform badge = Surface(outer, "RarityBadge", new Color(0.025f, 0.025f, 0.07f, 0.94f), 0.66f, 0.84f, 0.985f, 0.989f);
            Title(badge, card.Rarity.ToString(), compact ? (card.Rarity == CardRarity.SSR ? 24 : 28)
                : (card.Rarity == CardRarity.SSR ? 30 : 36), 0, 0, 1, 1).color = trim;
            if (isNew)
            {
                RectTransform ribbon = Surface(outer, "NewRibbon", new Color(0.72f, 0.09f, 0.055f), 0.018f, 0.885f, 0.30f, 0.988f);
                ribbon.localRotation = Quaternion.Euler(0, 0, 12);
                Title(ribbon, "NEW", 24, 0, 0, 1, 1);
            }
            if (clicked != null)
            {
                outline.raycastTarget = true;
                Button button = outer.gameObject.AddComponent<Button>();
                button.targetGraphic = outline;
                button.onClick.AddListener(() => clicked());
            }
            return outer;
        }

        private Button ActionButton(Transform parent, string text, float x0, float y0,
            float x1, float y1, Action action)
        {
            RectTransform rect = Surface(parent, text, theme.button, x0, y0, x1, y1);
            Image image = rect.GetComponent<Image>();
            image.raycastTarget = true;
            if (HasArt)
            {
                image.color = theme.primary;
                RectTransform inset = Rect(rect, "ButtonInset", 0, 0, 1, 1);
                RawImage fill = inset.gameObject.AddComponent<RawImage>();
                fill.texture = Texture2D.whiteTexture;
                fill.color = new Color(0.035f, 0.07f, 0.14f, 1);
                fill.raycastTarget = false;
                inset.offsetMin = new Vector2(2, 2);
                inset.offsetMax = new Vector2(-2, -2);
            }
            else if (theme.buttonNormal != null)
            {
                image.sprite = ButtonSprite(theme.buttonNormal);
                image.type = Image.Type.Sliced;
                image.color = Color.white;
            }
            Button button = rect.gameObject.AddComponent<Button>();
            button.targetGraphic = image;
            if (HasArt) button.targetGraphic = rect.Find("ButtonInset").GetComponent<RawImage>();
            ColorBlock colors = button.colors;
            colors.highlightedColor = new Color(1.15f, 1.15f, 1.15f);
            colors.pressedColor = new Color(0.75f, 0.85f, 1f);
            colors.disabledColor = new Color(0.42f, 0.46f, 0.54f, 0.85f);
            button.colors = colors;
            button.onClick.AddListener(() => action());
            Text caption = Label(rect, text, 28, 0.03f, 0.06f, 0.97f, 0.94f);
            if (HasArt) caption.font = titleFont;
            return button;
        }

            refillButton.onClick.Invoke();

        public IEnumerator RunUICheck(Action<string> completed, string folder, Action<int, int> resize)
        {
            var failures = new List<string>();
            int[] widths = { 1920, 1280, 1600 };
            int[] heights = { 1080, 720, 1000 };
            for (int size = 0; size < widths.Length; size++)
            {
                resize(widths[size], heights[size]);
                yield return null;
                yield return null;
                ShowSummon();
                yield return CaptureUI(folder, "summon", failures);
                OpenRules();
                yield return CaptureUI(folder, "rules", failures);
                CloseRules();
                int before = game.Currency;
                singleButton.onClick.Invoke();
                yield return CommitSigningForLegacyCheck();
                Summon(1); // 演出中的第二次请求不得扣费。
                bool locked = busy && !singleButton.interactable && game.Currency == before - 1;
                if (!locked) failures.Add("重复召唤锁定");
                yield return null;
                SkipReveal();
                if (lastResults.Count != 1 || currentScreen != "Summary") failures.Add("单抽");
                yield return CaptureUI(folder, "single", failures);
                summaryTenButton.onClick.Invoke();
                yield return CommitSigningForLegacyCheck();
                yield return null;
                SkipReveal();
                if (lastResults.Count != 10 || game.Currency != before - 11) failures.Add("十连/跳过");
                yield return CaptureUI(folder, "summary", failures);
                Button cardButton = summaryCards.GetComponentInChildren<Button>();
                cardButton.onClick.Invoke();
                if (currentScreen != "Detail") failures.Add("总览卡牌详情绑定");
                yield return CaptureUI(folder, "detail", failures);
                ToggleDetailUI();
                if (detailChrome.blocksRaycasts || !restoreUI.gameObject.activeSelf) failures.Add("隐藏界面");
                yield return CaptureUI(folder, "artwork", failures);
                restoreUI.onClick.Invoke();
                if (detailChrome.alpha != 1 || !header.gameObject.activeSelf) failures.Add("恢复界面");
                ShowScreen(detailReturn);
                OpenCollection();
                yield return CaptureUI(folder, "collection", failures);
                // 每种窗口使用相同足量资源，补充仅作用于独立 UI 场景。
                refillButton.onClick.Invoke();
            }
            while (game.Currency >= 10) { Summon(10); yield return CommitSigningForLegacyCheck(); SkipReveal(); }
            while (game.Currency > 0) { Summon(1); yield return CommitSigningForLegacyCheck(); SkipReveal(); }
            int pulls = game.TotalPulls;
            Summon(10);
            if (game.TotalPulls != pulls || tenButton.interactable) failures.Add("资源不足");
            yield return CaptureUI(folder, "insufficient", failures);
            refillButton.onClick.Invoke();
            if (game.Currency != 10) failures.Add("免费补充");
            Preview("UR001");
            yield return null;
            SkipReveal();
            if (game.Currency != 10 || game.TotalPulls != pulls) failures.Add("免费预览");
            completed(failures.Count == 0 ? "PASS: 单抽/十连/锁定/跳过/详情/隐藏恢复/收藏/不足/补充/预览/三种窗口布局/字体"
                : "FAIL: " + string.Join("; ", failures));
        }

        private IEnumerator CaptureUI(string folder, string page, List<string> failures)
        {
            yield return null;
            Canvas.ForceUpdateCanvases();
            yield return new WaitForEndOfFrame();
            foreach (Button button in GetComponentsInChildren<Button>())
            {
                if (!button.isActiveAndEnabled || !button.interactable) continue;
                var corners = new Vector3[4];
                button.GetComponent<RectTransform>().GetWorldCorners(corners);
                if (corners[0].x < -1 || corners[0].y < -1 || corners[2].x > Screen.width + 1
                    || corners[2].y > Screen.height + 1) failures.Add(page + ":按钮超出窗口:" + button.name);
            }
            foreach (Text text in GetComponentsInChildren<Text>())
            {
                if (!text.isActiveAndEnabled) continue;
                if (text.transform.parent.name == "RarityBadge" && text.preferredWidth > text.rectTransform.rect.width + 0.5f)
                    failures.Add(page + ":稀有度标识宽度不足:" + text.text);
                text.font.RequestCharactersInTexture(text.text, text.fontSize, text.fontStyle);
                foreach (char c in text.text)
                    if (!char.IsWhiteSpace(c) && !text.font.HasCharacter(c)) failures.Add("字体缺字:" + c);
            }
            if (page == "concept_layout_preview" && Screen.width == 1280)
            {
                var lines = new List<string>();
                foreach (Image graphic in GetComponentsInChildren<Image>())
                    if (graphic.name == "ButtonInset" || graphic.name == "Portrait" || graphic.name == "PortraitWindow" || graphic.name == "SSR001" || graphic.name == "返回召唤")
                        lines.Add(graphic.transform.parent.name + "/" + graphic.name + " color=" + graphic.color + " rect=" + graphic.rectTransform.rect
                            + " depth=" + graphic.canvasRenderer.absoluteDepth + " tint=" + graphic.canvasRenderer.GetColor() + " cull=" + graphic.canvasRenderer.cull + " material=" + graphic.material.name + " shader=" + graphic.material.shader.name + " texture=" + graphic.mainTexture.name);
                lines.Add("正文=" + font.name + "; 标题=" + titleFont.name);
                foreach (RawImage graphic in GetComponentsInChildren<RawImage>())
                {
                    Mesh mesh = graphic.canvasRenderer.GetMesh();
                    lines.Add("RAW " + graphic.transform.parent.name + "/" + graphic.name + " color=" + graphic.color + " mesh=" + mesh.vertexCount
                        + " firstColor=" + (mesh.vertexCount > 0 ? mesh.colors32[0].ToString() : "none")
                        + " uv=" + graphic.uvRect + " alpha=" + graphic.canvasRenderer.GetInheritedAlpha());
                }
                System.IO.File.WriteAllLines(System.IO.Path.Combine(folder, "visual_runtime.txt"), lines);
            }
            Canvas.ForceUpdateCanvases();
            yield return null;
            yield return new WaitForEndOfFrame();
            Texture2D shot = ScreenCapture.CaptureScreenshotAsTexture();
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(folder,
                Screen.width + "x" + Screen.height + "_" + page + ".png"), shot.EncodeToPNG());
            Destroy(shot);
        }
