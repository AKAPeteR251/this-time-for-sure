        private void Summon(int count)
        {
            if (busy || modal.gameObject.activeSelf) return;
            if (game.Currency < count)
            {
                ShowSummon();
                hint.text = "资源不足，点击右上角免费补充 +10。";
                return;
            }
            isPreview = false;
            busy = true;
            lastResults = game.Draw(count);
            UpdateResources();
            revealRoutine = StartCoroutine(Reveal());
        }

        private void Preview(string characterId)
        {
            if (busy) return;
            busy = true;
            isPreview = true;
            lastResults = new List<PrototypePull>
            {
                new PrototypePull { Card = game.Cards.Find(card => card.Id == characterId) }
            };
            revealRoutine = StartCoroutine(Reveal());
        }

        private IEnumerator Reveal()
        {
            ShowScreen("Reveal");
            refillButton.interactable = false;
            revealSpeed = 1;
            PrototypeCard highest = lastResults[0].Card;
            foreach (PrototypePull pull in lastResults)
                if (pull.Card.Rarity > highest.Rarity) highest = pull.Card;
            float duration = highest.Rarity == CardRarity.UR ? urRevealSeconds
                : highest.Rarity == CardRarity.SSR ? ssrRevealSeconds : normalRevealSeconds;
            PrototypePortrait binding = portraits.Find(item => item.characterId == highest.Id);
            Sprite portrait = highest.Id == "SSR001" && theme.summonCharacter != null
                ? theme.summonCharacter : binding == null ? null : binding.portrait;
            fastReveal.Begin(highest, portrait);
            float elapsed = 0;
            while (elapsed < duration)
            {
                elapsed += Time.deltaTime * revealSpeed;
                fastReveal.Animate(Mathf.Clamp01(elapsed / Mathf.Max(0.01f, duration)));
                yield return null;
            }
            fastReveal.Finish();
            revealRoutine = null;
            ShowSummary();
        }

        private void SkipReveal()
        {
            if (!busy || lastResults == null) return;
            if (revealRoutine != null) StopCoroutine(revealRoutine);
            revealRoutine = null;
            fastReveal.Finish();
            ShowSummary();
        }
        private void ShowSummary()
        {
            busy = false;
            refillButton.interactable = true;
            UpdateResources();
            Clear(summaryCards);
            summaryHeading.text = HasArt ? "召唤结果" : (isPreview ? "高级预览 · 不消耗资源 / 不计入收藏" : "召唤结果 · " + lastResults.Count + " 张");
            if (summarySubtitle != null) summarySubtitle.text = isPreview ? "角色预览" : lastResults.Count == 10 ? "十连召唤" : "单次召唤";
            if (lastResults.Count == 1)
            {
                PrototypePull pull = lastResults[0];
                Card(summaryCards, pull.Card, 0.40f, 0.01f, 0.60f, 0.97f,
                    () => OpenDetail(pull.Card, "Summary"), pull.IsNew);
            }
            else
            {
                for (int i = 0; i < lastResults.Count; i++)
                {
                    PrototypePull pull = lastResults[i];
                    float x = (i % 5) * 0.20f;
                    float y = i < 5 ? 0.51f : 0.01f;
                    Card(summaryCards, pull.Card, x + (HasArt ? 0.01f : 0.04f), y, x + (HasArt ? 0.19f : 0.16f), y + 0.47f,
                        () => OpenDetail(pull.Card, "Summary"), pull.IsNew);
                }
            }
            ShowScreen("Summary");
        }

                + (game.CountOf(card.Id) == 0 ? "预览 · 尚未收录\n" : "已收录 · 本次运行内收藏\n")

        private void OpenCollection()
        {
            Clear(collectionCards);
            collectionHeading.text = "本轮收藏 · " + game.CollectedCharacters + " / " + game.Cards.Count;
            for (int i = 0; i < game.Cards.Count; i++)
            {
                PrototypeCard card = game.Cards[i];
                float x = (i % 6) / 6f;
                float y = i < 6 ? 0.51f : 0.01f;
                bool owned = game.CountOf(card.Id) > 0;
                RectTransform tile = Card(collectionCards, card, x + 0.025f, y,
                    x + 0.14f, y + 0.47f,
                    owned ? (Action)(() => OpenDetail(card, "Collection")) : null, false);
                if (!owned)
                {
                    tile.gameObject.AddComponent<CanvasGroup>().alpha = 0.24f;
                    Label(tile, "未收录", 21, 0.10f, 0.35f, 0.90f, 0.60f);
                }
            }
            ShowScreen("Collection");
        }

            foreach (var screen in screens) screen.Value.SetActive(screen.Key == name);

            {

                navigationUnderline.anchorMax = new Vector2(collection ? 0.59f : 0.495f, 0.20f);

            if (EventSystem.current != null) EventSystem.current.SetSelectedGameObject(null);

        private void ToggleDetailUI()
        {
            bool show = detailChrome.alpha < 0.5f;
            detailChrome.alpha = show ? 1 : 0;
            detailChrome.interactable = detailChrome.blocksRaycasts = show;
            header.gameObject.SetActive(show);
            if (sidebar != null) sidebar.gameObject.SetActive(show);
            restoreUI.gameObject.SetActive(!show);

        private RectTransform SpriteSurface(Transform parent, string name, Sprite sprite,
            float x0, float y0, float x1, float y1, bool preserve)
        {
            // Fitter 会接管其对象的锚点，因此先用独立 Slot 保留调用者定义的区域。
            RectTransform slot = preserve && sprite != null ? Rect(parent, name + "Slot", x0, y0, x1, y1) : null;
            RectTransform rect = slot != null ? Surface(slot, name, Color.white, 0, 0, 1, 1)
                : Surface(parent, name, Color.white, x0, y0, x1, y1);
            Image image = rect.GetComponent<Image>();
            image.sprite = sprite;
            if (preserve && sprite != null)
            {
                var fit = rect.gameObject.AddComponent<AspectRatioFitter>();
                fit.aspectMode = AspectRatioFitter.AspectMode.FitInParent;
                fit.aspectRatio = sprite.rect.width / sprite.rect.height;
            }
            return rect;
        }

        private Sprite ButtonSprite(Sprite source)
        {
            Sprite cropped;
            if (croppedButtons.TryGetValue(source, out cropped)) return cropped;
            // 使用同一底图裁掉透明留白；不修改 PNG，不做 SpriteSwap。
            Rect pixels = source == theme.buttonPrimary ? new Rect(20, 35, 792, 210) : new Rect(20, 31, 792, 210);
            cropped = Sprite.Create(source.texture, pixels, new Vector2(0.5f, 0.5f), 100, 0,
                SpriteMeshType.FullRect, new Vector4(90, 40, 90, 40));
            cropped.name = source.name + "_VisibleBody";
            croppedButtons.Add(source, cropped);
            return cropped;
        }

        private void OnDestroy()
        {
            foreach (Sprite sprite in croppedButtons.Values) Destroy(sprite);
        }

        private RectTransform Rect(Transform parent, string name, float x0, float y0, float x1, float y1)
        {
            var element = new GameObject(name, typeof(RectTransform));

            var rect = element.GetComponent<RectTransform>();
            rect.anchorMin = new Vector2(x0, y0);
            rect.anchorMax = new Vector2(x1, y1);
            rect.offsetMin = rect.offsetMax = Vector2.zero;
            return rect;
        }

        {
            var failures = new List<string>();
            if (useFixedMock || !HasArt || portraits.Count != 6) failures.Add("必须为六卡真实内存入口");
            if (FindObjectsOfType<EventSystem>().Length != 1 || GetComponentsInChildren<Canvas>().Length != 1)

            summaryTenButton.onClick.Invoke();

            int totalBeforeSkip = game.TotalPulls;

            SkipReveal();
            if (game.Currency != 19 || lastResults.Count != 10 || lastResults != calculated
                || game.TotalPulls != totalBeforeSkip) failures.Add("十连19/跳过固定结果");
            var seen = new HashSet<string>();

                if (!seen.Add(pull.Card.Id) && pull.IsNew) failures.Add("重复NEW:" + pull.Card.Id);

            if (detailChrome.alpha != 1 || !detailChrome.blocksRaycasts) failures.Add("隐藏恢复");
            OpenCollection();
            if (game.CollectedCharacters == 0) failures.Add("收藏");
            ShowSummon();
            refillButton.onClick.Invoke();
            if (game.Currency != 29 || game.TotalPulls != 11) failures.Add("补充29");
            int[] widths = { 1920, 1280, 1600 };
            int[] heights = { 1080, 720, 1000 };
            for (int i = 0; i < widths.Length; i++)
            {
                resize(widths[i], heights[i]);
                yield return null;
                yield return null;
                ShowSummon();
                yield return CaptureUI(folder, "summon", failures);
                OpenRules();
                yield return CaptureUI(folder, "rules", failures);
                CloseRules();
                lastResults = calculated;
                isPreview = false;
                ShowSummary();
                yield return CaptureUI(folder, "summary_random", failures);
                OpenDetail(game.Cards.Find(card => card.Id == "SSR001"), "Summary");
                yield return CaptureUI(folder, "detail_SSR001", failures);
                ToggleDetailUI();
                yield return CaptureUI(folder, "artwork_SSR001", failures);
                restoreUI.onClick.Invoke();
                OpenDetail(game.Cards.Find(card => card.Id == "UR001"), "Summary");

                    for (int n = 0; n < portraits.Count; n++)

                        float y = n < 3 ? 0.48f : 0;
                        Card(calibration, definition, x + 0.035f, y, x + 0.30f, y + 0.44f, null, false);

                ShowScreen("ArtCalibration");

            completed(failures.Count == 0 ? "PASS: 真实内存30→29→19→29; 单抽/十连/跳过固定结果/NEW/详情/隐藏恢复/收藏/SSR与UR预览/六卡图/三种窗口/字体"
                : "FAIL: " + string.Join("; ", failures));
        }

        public IEnumerator RunUICheck(Action<string> completed, string folder, Action<int, int> resize)
        {
            var failures = new List<string>();
            int[] widths = { 1920, 1280, 1600 };
            int[] heights = { 1080, 720, 1000 };
            for (int size = 0; size < widths.Length; size++)
            {
                resize(widths[size], heights[size]);
                yield return null;
                yield return null;
                ShowSummon();
                yield return CaptureUI(folder, "summon", failures);
                OpenRules();
                yield return CaptureUI(folder, "rules", failures);
                CloseRules();
                int before = game.Currency;
                singleButton.onClick.Invoke();
                Summon(1); // 演出中的第二次请求不得扣费。
                bool locked = busy && !singleButton.interactable && game.Currency == before - 1;
                if (!locked) failures.Add("重复召唤锁定");
                yield return null;
                SkipReveal();
                if (lastResults.Count != 1 || currentScreen != "Summary") failures.Add("单抽");
                yield return CaptureUI(folder, "single", failures);
                summaryTenButton.onClick.Invoke();
                yield return null;
                SkipReveal();
                if (lastResults.Count != 10 || game.Currency != before - 11) failures.Add("十连/跳过");
                yield return CaptureUI(folder, "summary", failures);
                Button cardButton = summaryCards.GetComponentInChildren<Button>();
                cardButton.onClick.Invoke();
                if (currentScreen != "Detail") failures.Add("总览卡牌详情绑定");
                yield return CaptureUI(folder, "detail", failures);
                ToggleDetailUI();
                if (detailChrome.blocksRaycasts || !restoreUI.gameObject.activeSelf) failures.Add("隐藏界面");
                yield return CaptureUI(folder, "artwork", failures);
                restoreUI.onClick.Invoke();
                if (detailChrome.alpha != 1 || !header.gameObject.activeSelf) failures.Add("恢复界面");
                ShowScreen(detailReturn);
                OpenCollection();
                yield return CaptureUI(folder, "collection", failures);
                // 每种窗口使用相同足量资源，补充仅作用于独立 UI 场景。
                refillButton.onClick.Invoke();
            }
            while (game.Currency >= 10) { Summon(10); SkipReveal(); }
            while (game.Currency > 0) { Summon(1); SkipReveal(); }
            int pulls = game.TotalPulls;
            Summon(10);
            if (game.TotalPulls != pulls || !hint.text.Contains("资源不足")) failures.Add("资源不足");
            yield return CaptureUI(folder, "insufficient", failures);
            refillButton.onClick.Invoke();
            if (game.Currency != 10) failures.Add("免费补充");
            Preview("UR001");
            yield return null;
            SkipReveal();
            if (game.Currency != 10 || game.TotalPulls != pulls) failures.Add("免费预览");

                : "FAIL: " + string.Join("; ", failures));
        }

        private IEnumerator CaptureUI(string folder, string page, List<string> failures)
        {
            yield return null;
            Canvas.ForceUpdateCanvases();
            yield return new WaitForEndOfFrame();
            foreach (Button button in GetComponentsInChildren<Button>())
            {
                if (!button.isActiveAndEnabled || !button.interactable) continue;
                var corners = new Vector3[4];
                button.GetComponent<RectTransform>().GetWorldCorners(corners);
                if (corners[0].x < -1 || corners[0].y < -1 || corners[2].x > Screen.width + 1
                    || corners[2].y > Screen.height + 1) failures.Add(page + ":按钮超出窗口:" + button.name);
            }
            foreach (Text text in GetComponentsInChildren<Text>())
            {
                if (!text.isActiveAndEnabled) continue;
                if (text.transform.parent.name == "RarityBadge" && text.preferredWidth > text.rectTransform.rect.width + 0.5f)
                    failures.Add(page + ":稀有度标识宽度不足:" + text.text);
                text.font.RequestCharactersInTexture(text.text, text.fontSize, text.fontStyle);
                foreach (char c in text.text)
                    if (!char.IsWhiteSpace(c) && !text.font.HasCharacter(c)) failures.Add("字体缺字:" + c);
            }
            if (page == "concept_layout_preview" && Screen.width == 1280)
            {
                var lines = new List<string>();
                foreach (Image graphic in GetComponentsInChildren<Image>())
                    if (graphic.name == "ButtonInset" || graphic.name == "Portrait" || graphic.name == "PortraitWindow" || graphic.name == "SSR001" || graphic.name == "返回召唤")
                        lines.Add(graphic.transform.parent.name + "/" + graphic.name + " color=" + graphic.color + " rect=" + graphic.rectTransform.rect
                            + " depth=" + graphic.canvasRenderer.absoluteDepth + " tint=" + graphic.canvasRenderer.GetColor() + " cull=" + graphic.canvasRenderer.cull + " material=" + graphic.material.name + " shader=" + graphic.material.shader.name + " texture=" + graphic.mainTexture.name);
                lines.Add("正文=" + font.name + "; 标题=" + titleFont.name);
                foreach (RawImage graphic in GetComponentsInChildren<RawImage>())
                {
                    Mesh mesh = graphic.canvasRenderer.GetMesh();
                    lines.Add("RAW " + graphic.transform.parent.name + "/" + graphic.name + " color=" + graphic.color + " mesh=" + mesh.vertexCount
                        + " firstColor=" + (mesh.vertexCount > 0 ? mesh.colors32[0].ToString() : "none")
                        + " uv=" + graphic.uvRect + " alpha=" + graphic.canvasRenderer.GetInheritedAlpha());
                }
                System.IO.File.WriteAllLines(System.IO.Path.Combine(folder, "visual_runtime.txt"), lines);
            }
            Canvas.ForceUpdateCanvases();
            yield return null;
            yield return new WaitForEndOfFrame();
            Texture2D shot = ScreenCapture.CaptureScreenshotAsTexture();
            System.IO.File.WriteAllBytes(System.IO.Path.Combine(folder,
                Screen.width + "x" + Screen.height + "_" + page + ".png"), shot.EncodeToPNG());
            Destroy(shot);
        }

        {

            yield return null;
            yield return null;

                + "; 十连及跳过=" + tenPassed + "; 收藏=" + collectionPassed;

        }
